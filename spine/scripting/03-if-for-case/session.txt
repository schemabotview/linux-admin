#!/bin/bash
# check-paths: ask the site for each path; fail if any answer is unexpected
failed=0
for path in / /status.txt /about.html /missing; do
    code=$(curl -s -o /dev/null -w '%{http_code}' "https://sysops.example$path")
    case $code in
        200) echo "ok       $path" ;;
        404) echo "missing  $path" ;;
        *)   echo "ERROR    $path ($code)"; failed=$((failed + 1)) ;;
    esac
done
if [ "$failed" -gt 0 ]; then
    echo "$failed path(s) failed"
    exit 1
fi
echo "all paths answered"

exit 0
